SELECT 
		s.sid,
    s.aname AS "Schuldner",
    to_char(
        COALESCE(SUM(vb.betrag) FILTER (WHERE vb.konto_zu_sid = 4), 0) -
        COALESCE(SUM(vb.betrag) FILTER (WHERE vb.konto_von_sid = 4), 0),
        'FM€ 999G999G990D00'
    ) AS "Schuldenstand"
FROM schuldner s
left JOIN transaktion t ON t.fk_schuldner_sid = s.sid AND t.fk_recordstate_sid = 0
left JOIN v_bewegungen vb ON vb.source_sid = t.sid AND vb.source_typ = 'transaktion'
WHERE s.fk_recordstate_sid = 0 and s.validto ISNULL or s.validto > now()
GROUP BY s.sid, s.aname
ORDER BY "Schuldenstand" DESC;
