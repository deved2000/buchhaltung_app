WITH basis AS (
  SELECT
      kategorie,
      coalesce(farbe, '#999999')       AS farbe,
      coalesce(betrag, 0)              AS monat,
      budget,
      coalesce(schnitt_12m, 0)         AS schnitt
  FROM v_kategorie_monat_vergleich
  WHERE monat = date_trunc('month', {{ (DatePicker_month.selectedDate || new Date().toISOString()) }}::timestamptz AT TIME ZONE 'Europe/Berlin')::date
    AND (betrag > 0 OR budget IS NOT NULL)
),
zeilen AS (
  SELECT 0 AS sortier, kategorie, farbe, monat, budget, schnitt FROM basis
  UNION ALL
  SELECT 1, 'Gesamt', '#455a64', sum(monat), sum(budget), sum(schnitt) FROM basis
)
SELECT
    kategorie                                          AS "Kategorie",
    farbe                                              AS "Farbe",
    monat::float8                                      AS "Monat",
    budget::float8                                     AS "Budget",
    schnitt::float8                                    AS "Schnitt",
    CASE WHEN budget IS NULL OR budget = 0 THEN 0
         ELSE round(100.0 * monat / budget)::int END   AS "BudgetProzent",
    (monat - schnitt)::float8                          AS "DiffSchnitt",
    (sortier = 1)                                      AS "IstGesamt"
FROM zeilen
ORDER BY sortier, monat DESC;
