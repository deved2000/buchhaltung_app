SELECT
    konto_name AS "Konto",
	to_char(saldo_anfang, 'FM€ 999G999G990D00') AS "Anfangssaldo",
	to_char(saldo_ende,   'FM€ 999G999G990D00') AS "akt. Saldo",
	konto_sid
FROM v_monatssalden
WHERE monat = date_trunc('month', {{DatePicker_month.selectedDate}}::timestamptz AT TIME ZONE 'Europe/Berlin')::date

UNION ALL

SELECT
    'Gesamt' AS "Konto",
    to_char(SUM(saldo_anfang), 'FM€ 999G999G990D00') AS "Anfangssaldo",
    to_char(SUM(saldo_ende),   'FM€ 999G999G990D00') AS "akt. Saldo",
    999999 AS konto_sid
FROM v_monatssalden
WHERE monat = date_trunc('month', {{DatePicker_month.selectedDate}}::timestamptz AT TIME ZONE 'Europe/Berlin')::date

order by konto_sid
