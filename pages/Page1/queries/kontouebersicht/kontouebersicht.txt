SELECT * FROM (
    SELECT
        k.aname AS "Konto",
        to_char(
            COALESCE(SUM(
                CASE
                    WHEN vb.konto_zu_sid = k.sid THEN vb.betrag
                    WHEN vb.konto_von_sid = k.sid THEN -vb.betrag
                    ELSE 0
                END
            ), 0),
            'FM€ 999G999G990D00'
        ) AS "Saldo"
    FROM konto k
    LEFT JOIN v_bewegungen vb
        ON vb.konto_zu_sid = k.sid
        OR vb.konto_von_sid = k.sid
    WHERE k.fk_recordstate_sid = 0
    GROUP BY k.sid, k.aname

    UNION ALL

    SELECT
        'Gesamt' AS "Konto",
        to_char(
            COALESCE(SUM(
                CASE
                    WHEN vb.konto_zu_sid = k.sid THEN vb.betrag
                    WHEN vb.konto_von_sid = k.sid THEN -vb.betrag
                    ELSE 0
                END
            ), 0),
            'FM€ 999G999G990D00'
        ) AS "Saldo"
    FROM konto k
    LEFT JOIN v_bewegungen vb
        ON vb.konto_zu_sid = k.sid
        OR vb.konto_von_sid = k.sid
    WHERE k.fk_recordstate_sid = 0
) sub
ORDER BY "Konto" = 'Gesamt', "Konto";
